-- 01 Труппы из города
PARAMETERS [Введите город] Text;
SELECT ШифрТруппы, НазваниеТеатра, КоличествоЧеловек, КоличествоСпектаклей
FROM [ТеатральныеТруппы]
WHERE Город = [Введите город]
ORDER BY НазваниеТеатра;

-- 02 Концертные залы города
PARAMETERS [Введите город] Text;
SELECT ШифрЗала, Адрес, КоличествоМест, СтоимостьБилета
FROM [КонцертныеЗалы]
WHERE Город = [Введите город]
ORDER BY КоличествоМест DESC;

-- 03 Гастрольный график труппы
PARAMETERS [Введите шифр труппы] Text;
SELECT В.НазваниеСпектакля, В.Дата, В.Время, З.Город, З.Адрес
FROM [Выступления] AS [В] INNER JOIN [КонцертныеЗалы] AS [З] ON В.ШифрЗала = З.ШифрЗала
WHERE В.ШифрТруппы = [Введите шифр труппы]
ORDER BY В.Дата, В.Время;

-- 04 Афиша зала за период
PARAMETERS [Введите шифр зала] Text, [Начальная дата] DateTime, [Конечная дата] DateTime;
SELECT В.Дата, В.Время, Т.НазваниеТеатра, В.НазваниеСпектакля
FROM [Выступления] AS [В] INNER JOIN [ТеатральныеТруппы] AS [Т] ON В.ШифрТруппы = Т.ШифрТруппы
WHERE В.ШифрЗала = [Введите шифр зала] AND В.Дата Between [Начальная дата] And [Конечная дата]
ORDER BY В.Дата, В.Время;

-- 05 Спектакли в городе
PARAMETERS [Введите город] Text;
SELECT В.Дата, В.Время, В.НазваниеСпектакля, Т.НазваниеТеатра, З.Адрес
FROM ([Выступления] AS [В] INNER JOIN [ТеатральныеТруппы] AS [Т] ON В.ШифрТруппы = Т.ШифрТруппы) INNER JOIN [КонцертныеЗалы] AS [З] ON В.ШифрЗала = З.ШифрЗала
WHERE З.Город = [Введите город]
ORDER BY В.Дата, В.Время;

-- 06 Где идёт спектакль
PARAMETERS [Введите название спектакля] Text;
SELECT В.НазваниеСпектакля, Т.НазваниеТеатра, З.Город, В.Дата, В.Время
FROM ([Выступления] AS [В] INNER JOIN [ТеатральныеТруппы] AS [Т] ON В.ШифрТруппы = Т.ШифрТруппы) INNER JOIN [КонцертныеЗалы] AS [З] ON В.ШифрЗала = З.ШифрЗала
WHERE В.НазваниеСпектакля Like "*" & [Введите название спектакля] & "*"
ORDER BY В.Дата;

-- 07 Залы вместимостью не менее
PARAMETERS [Введите минимальное количество мест] Long;
SELECT ШифрЗала, Город, Адрес, КоличествоМест
FROM [КонцертныеЗалы]
WHERE КоличествоМест >= [Введите минимальное количество мест]
ORDER BY КоличествоМест DESC;

-- 08 Выручка от выступлений
SELECT Т.НазваниеТеатра, В.НазваниеСпектакля, З.Город, В.Дата, В.ПроданоБилетов, З.СтоимостьБилета, [В].[ПроданоБилетов]*[З].[СтоимостьБилета] AS [Выручка]
FROM ([Выступления] AS [В] INNER JOIN [ТеатральныеТруппы] AS [Т] ON В.ШифрТруппы = Т.ШифрТруппы) INNER JOIN [КонцертныеЗалы] AS [З] ON В.ШифрЗала = З.ШифрЗала
ORDER BY В.Дата;

-- 09 Заполняемость залов
SELECT З.Город, В.НазваниеСпектакля, В.Дата, З.КоличествоМест, В.ПроданоБилетов, [З].[КоличествоМест]-[В].[ПроданоБилетов] AS [Свободных мест], Round([В].[ПроданоБилетов]/[З].[КоличествоМест]*100,1) AS [Заполняемость в процентах]
FROM [Выступления] AS [В] INNER JOIN [КонцертныеЗалы] AS [З] ON В.ШифрЗала = З.ШифрЗала
ORDER BY Round([В].[ПроданоБилетов]/[З].[КоличествоМест]*100,1) DESC;

-- 10 Итоги гастролей по труппам
SELECT Т.НазваниеТеатра, Count(В.НазваниеСпектакля) AS [Число выступлений], Sum(В.ПроданоБилетов) AS [Продано билетов], Sum([В].[ПроданоБилетов]*[З].[СтоимостьБилета]) AS [Выручка]
FROM ([Выступления] AS [В] INNER JOIN [ТеатральныеТруппы] AS [Т] ON В.ШифрТруппы = Т.ШифрТруппы) INNER JOIN [КонцертныеЗалы] AS [З] ON В.ШифрЗала = З.ШифрЗала
GROUP BY Т.НазваниеТеатра
ORDER BY Sum([В].[ПроданоБилетов]*[З].[СтоимостьБилета]) DESC;

-- 11 Статистика по залам
SELECT З.ШифрЗала, З.Город, Count(В.Дата) AS [Число выступлений], Avg(В.ПроданоБилетов) AS [В среднем продано], Max(В.ПроданоБилетов) AS [Максимум продано], Min(В.ПроданоБилетов) AS [Минимум продано]
FROM [КонцертныеЗалы] AS [З] INNER JOIN [Выступления] AS [В] ON З.ШифрЗала = В.ШифрЗала
GROUP BY З.ШифрЗала, З.Город;

-- 12 Пять самых посещаемых выступлений
SELECT TOP 5 Т.НазваниеТеатра, В.НазваниеСпектакля, З.Город, В.Дата, В.ПроданоБилетов
FROM ([Выступления] AS [В] INNER JOIN [ТеатральныеТруппы] AS [Т] ON В.ШифрТруппы = Т.ШифрТруппы) INNER JOIN [КонцертныеЗалы] AS [З] ON В.ШифрЗала = З.ШифрЗала
ORDER BY В.ПроданоБилетов DESC;

-- 13 Выступления с низкой заполняемостью
PARAMETERS [Введите порог заполняемости в процентах] IEEEDouble;
SELECT Т.НазваниеТеатра, В.НазваниеСпектакля, З.Город, В.Дата, Round([В].[ПроданоБилетов]/[З].[КоличествоМест]*100,1) AS [Заполняемость в процентах]
FROM ([Выступления] AS [В] INNER JOIN [ТеатральныеТруппы] AS [Т] ON В.ШифрТруппы = Т.ШифрТруппы) INNER JOIN [КонцертныеЗалы] AS [З] ON В.ШифрЗала = З.ШифрЗала
WHERE [В].[ПроданоБилетов]/[З].[КоличествоМест]*100 < [Введите порог заполняемости в процентах]
ORDER BY В.Дата;

-- 14 Гастроли вне родного города
SELECT DISTINCT Т.НазваниеТеатра, Т.Город AS [Город труппы], З.Город AS [Город гастролей]
FROM ([Выступления] AS [В] INNER JOIN [ТеатральныеТруппы] AS [Т] ON В.ШифрТруппы = Т.ШифрТруппы) INNER JOIN [КонцертныеЗалы] AS [З] ON В.ШифрЗала = З.ШифрЗала
WHERE Т.Город <> З.Город
ORDER BY Т.НазваниеТеатра;

-- 15 Стоимость билета по категориям мест
PARAMETERS [Введите шифр зала] Text;
SELECT З.Город, З.Адрес, К.КатегорияМест, К.РядС, К.РядПо, З.СтоимостьБилета AS [Базовая стоимость], К.Коэффициент, CCur([З].[СтоимостьБилета]*[К].[Коэффициент]) AS [Стоимость билета]
FROM [КонцертныеЗалы] AS [З] INNER JOIN [КатегорииМест] AS [К] ON З.ШифрЗала = К.ШифрЗала
WHERE З.ШифрЗала = [Введите шифр зала]
ORDER BY К.Коэффициент DESC;

-- 16 Залы по городам
SELECT Город, Count(ШифрЗала) AS [Число залов], Sum(КоличествоМест) AS [Всего мест], Avg(СтоимостьБилета) AS [Средняя стоимость билета]
FROM [КонцертныеЗалы]
GROUP BY Город
ORDER BY Город;

-- 17 Перекрёстный - билеты по городам
TRANSFORM Sum(В.ПроданоБилетов) AS [Продано]
SELECT Т.НазваниеТеатра AS [Труппа]
FROM ([Выступления] AS [В] INNER JOIN [ТеатральныеТруппы] AS [Т] ON В.ШифрТруппы = Т.ШифрТруппы) INNER JOIN [КонцертныеЗалы] AS [З] ON В.ШифрЗала = З.ШифрЗала
GROUP BY Т.НазваниеТеатра
PIVOT З.Город;

-- 18 Перекрёстный - выступления по месяцам
TRANSFORM Count(В.НазваниеСпектакля) AS [Выступлений]
SELECT Т.НазваниеТеатра AS [Труппа]
FROM [ТеатральныеТруппы] AS [Т] INNER JOIN [Выступления] AS [В] ON Т.ШифрТруппы = В.ШифрТруппы
GROUP BY Т.НазваниеТеатра
PIVOT Format(В.Дата,"yyyy-mm");

-- 19 Труппы с числом выступлений не менее
PARAMETERS [Введите минимальное число выступлений] Long;
SELECT Т.ШифрТруппы, Т.НазваниеТеатра, Count(В.ШифрЗала) AS [Число выступлений]
FROM [ТеатральныеТруппы] AS [Т] INNER JOIN [Выступления] AS [В] ON Т.ШифрТруппы = В.ШифрТруппы
GROUP BY Т.ШифрТруппы, Т.НазваниеТеатра
HAVING Count(В.ШифрЗала) >= [Введите минимальное число выступлений]
ORDER BY Count(В.ШифрЗала) DESC;

-- 20 Предстоящие выступления
SELECT В.Дата, В.Время, З.Город, Т.НазваниеТеатра, В.НазваниеСпектакля, DateDiff("d",Date(),[В].[Дата]) AS [Дней до начала]
FROM ([Выступления] AS [В] INNER JOIN [ТеатральныеТруппы] AS [Т] ON В.ШифрТруппы = Т.ШифрТруппы) INNER JOIN [КонцертныеЗалы] AS [З] ON В.ШифрЗала = З.ШифрЗала
WHERE В.Дата >= Date()
ORDER BY В.Дата, В.Время;

-- 21 Изменение стоимости билета
PARAMETERS [Введите шифр зала] Text, [Введите новую стоимость билета] Currency;
UPDATE [КонцертныеЗалы]
SET [СтоимостьБилета] = [Введите новую стоимость билета]
WHERE ШифрЗала = [Введите шифр зала];

-- 22 Повышение цен в городе на процент
PARAMETERS [Введите город] Text, [Введите процент повышения] IEEEDouble;
UPDATE [КонцертныеЗалы]
SET [СтоимостьБилета] = [СтоимостьБилета]*(1+[Введите процент повышения]/100)
WHERE Город = [Введите город];

-- 23 Продажа билетов на выступление
PARAMETERS [Введите шифр зала] Text, [Введите дату] DateTime, [Введите время] DateTime, [Введите количество проданных билетов] Long;
UPDATE [Выступления]
SET [ПроданоБилетов] = [ПроданоБилетов]+[Введите количество проданных билетов]
WHERE ШифрЗала = [Введите шифр зала] AND Дата = [Введите дату] AND Время = [Введите время];

-- 24 Добавление труппы
PARAMETERS [Введите шифр труппы] Text, [Введите название театра] Text, [Введите город] Text, [Введите количество человек] Long, [Введите количество спектаклей] Long;
INSERT INTO [ТеатральныеТруппы] ([ШифрТруппы], [НазваниеТеатра], [Город], [КоличествоЧеловек], [КоличествоСпектаклей])
SELECT [Введите шифр труппы] AS [Ш], [Введите название театра] AS [Н], [Введите город] AS [Г], [Введите количество человек] AS [Ч], [Введите количество спектаклей] AS [С];

-- 25 Добавление выступления
PARAMETERS [Введите шифр труппы] Text, [Введите шифр зала] Text, [Введите название спектакля] Text, [Введите дату] DateTime, [Введите время] DateTime;
INSERT INTO [Выступления] ([ШифрТруппы], [ШифрЗала], [НазваниеСпектакля], [Дата], [Время], [ПроданоБилетов])
SELECT [Введите шифр труппы] AS [Т], [Введите шифр зала] AS [З], [Введите название спектакля] AS [С], [Введите дату] AS [Д], [Введите время] AS [В], 0 AS [П];

-- 26 Удаление выступления
PARAMETERS [Введите шифр зала] Text, [Введите дату] DateTime, [Введите время] DateTime;
DELETE *
FROM [Выступления]
WHERE ШифрЗала = [Введите шифр зала] AND Дата = [Введите дату] AND Время = [Введите время];

-- 27 Удаление выступлений до даты
PARAMETERS [Введите дату] DateTime;
DELETE *
FROM [Выступления]
WHERE Дата < [Введите дату];

-- Форма_Выступления труппы
SELECT В.ШифрТруппы, В.Дата, В.Время, В.НазваниеСпектакля, З.Город, З.Адрес, В.ПроданоБилетов
FROM [Выступления] AS [В] INNER JOIN [КонцертныеЗалы] AS [З] ON В.ШифрЗала = З.ШифрЗала
ORDER BY В.Дата, В.Время;

-- Форма_Афиша зала
SELECT В.ШифрЗала, В.Дата, В.Время, Т.НазваниеТеатра, В.НазваниеСпектакля, В.ПроданоБилетов
FROM [Выступления] AS [В] INNER JOIN [ТеатральныеТруппы] AS [Т] ON В.ШифрТруппы = Т.ШифрТруппы
ORDER BY В.Дата, В.Время;

-- Форма_Цены по категориям
SELECT К.ШифрЗала, К.КатегорияМест, К.РядС, К.РядПо, К.Коэффициент, CCur([З].[СтоимостьБилета]*[К].[Коэффициент]) AS [Стоимость билета]
FROM [КатегорииМест] AS [К] INNER JOIN [КонцертныеЗалы] AS [З] ON К.ШифрЗала = З.ШифрЗала
ORDER BY К.Коэффициент DESC;
